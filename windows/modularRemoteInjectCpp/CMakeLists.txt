macro(require_exactly_one PHASE)
    set(_count 0)
    foreach(_opt ${ARGN})
        if(${_opt})
            math(EXPR _count "${_count} + 1")
        endif()
    endforeach()
    if(_count GREATER 1)
        message(FATAL_ERROR "${PHASE}: multiple variants enabled (${ARGN}) — pick exactly one")
    endif()
    if(_count EQUAL 0)
        message(FATAL_ERROR "${PHASE}: no variant enabled (${ARGN}) — pick exactly one")
    endif()
endmacro()
